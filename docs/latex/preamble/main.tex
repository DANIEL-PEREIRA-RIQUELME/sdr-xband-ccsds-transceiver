% ********************************************************************
% Entorno para código C++
\lstdefinestyle{styleCPP}{
    language=C++,
    escapeinside={\%*}{*)},          % permite añadir LaTeX dentro del código
    basicstyle=\linespread{1}\ttfamily\footnotesize,
    xleftmargin=40pt, 
    xrightmargin=0pt,
    frame=leftline,
    framesep=9pt,
    framerule=2pt,
    rulecolor=\color{orange},        % Borde izquierdo en naranja
    numbers=left,
    numberstyle=\scriptsize\ttfamily\color{gray},
    numbersep=20pt,
    breakautoindent=true,
    breakindent=20pt,
    keywordstyle=\color{blue},
    stringstyle=\color{red},
    commentstyle=\color{green!50!black}, % Verde estándar sin depender de "mygreen"
    showstringspaces=false,
    upquote=true,
    backgroundcolor=\color{white}    % Fondo blanco independiente
}

% Comando para invocar el entorno C++
\lstnewenvironment{cpp}[1][1]{\lstset{style=styleCPP, firstnumber=#1}}{}

% ********************************************************************
